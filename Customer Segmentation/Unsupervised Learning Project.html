from scipy.spatial.distance import cdist
clusters=range(1,10)
meanDistortions=[]

for k in clusters:
    model=KMeans(n_clusters=k)
    model.fit(scaled_d)
    prediction=model.predict(scaled_d)
    meanDistortions.append(sum(np.min(cdist(scaled_d, model.cluster_centers_, 'euclidean'), axis=1)) / scaled_d.shape[0])


plt.plot(clusters, meanDistortions, 'bx-')
plt.xlabel('k')
plt.ylabel('Average distortion')
plt.title('Selecting k with the Elbow Method')
